const express = require('express');
const cors = require('cors');
const multer = require('multer');
const path = require('path');
const fs = require('fs');

const app = express();
const PORT = 3000;

app.use(cors());
app.use(express.json());

// ==========================================
// 1. FILE UPLOAD CONFIGURATION (MULTER)
// ==========================================
// Create an 'uploads' directory if it doesn't exist
const uploadDir = path.join(__dirname, 'uploads');
if (!fs.existsSync(uploadDir)) {
  fs.mkdirSync(uploadDir);
}

// Configure where and how to save the uploaded photos
const storage = multer.diskStorage({
  destination: function (req, file, cb) {
    cb(null, uploadDir);
  },
  filename: function (req, file, cb) {
    // Save file with a timestamp to prevent naming collisions
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
    cb(null, 'insureai-' + uniqueSuffix + path.extname(file.originalname));
  }
});

const upload = multer({ 
  storage: storage,
  limits: { fileSize: 20 * 1024 * 1024 } // 20MB limit per file
});

// Serve the uploads folder publicly so the frontend can preview the images
app.use('/uploads', express.static(uploadDir));


// ==========================================
// 2. MOCK DATABASE
// ==========================================

    { id: "#INS-8821", customer: "Arjun Mehta", type: "Motor – Collision", amount: "₹42,000", fraudScore: "3%", fraudLevel: "low", status: "Approved" }
  ]
};


// ==========================================
// 3. API ENDPOINTS
// ==========================================

// Endpoint to handle the raw image file uploads
app.post('/api/upload', upload.array('photos', 10), (req, res) => {
  if (!req.files || req.files.length === 0) {
    return res.status(400).json({ error: 'No files uploaded.' });
  }

  // Generate public URLs for the uploaded files
  const fileUrls = req.files.map(file => `http://localhost:${PORT}/uploads/${file.filename}`);
  
  console.log(`📸 Received ${req.files.length} photos for processing.`);
  res.status(200).json({ message: 'Upload successful', urls: fileUrls });
});

// Endpoint to save the finalized claim
app.post('/api/claims', (req, res) => {
  const { customer, type, amount, fraudScore } = req.body;
  
  // Create a new claim object
  const newClaim = {
    id: `#INS-${8822 + dashboardData.claims.length}`,
    customer: customer || "Unknown Customer",
    type: type || "Unspecified",
    amount: `₹${parseInt(amount.replace(/,/g, '')).toLocaleString()}`,
    fraudScore: `${fraudScore}%`,
    fraudLevel: fraudScore < 25 ? 'low' : fraudScore < 60 ? 'med' : 'high',
    status: fraudScore > 60 ? "Flagged" : "In Review"
  };

  // Add to database
  dashboardData.claims.unshift(newClaim);
  dashboardData.kpis.totalClaims += 1;
  if (newClaim.status === "Flagged") dashboardData.kpis.fraudFlags += 1;

  console.log(`💾 Saved new claim: ${newClaim.id} for ${newClaim.customer}`);
  res.status(201).json({ message: "Claim submitted successfully", claim: newClaim });
});

// Start the server
