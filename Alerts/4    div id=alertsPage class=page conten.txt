### Step 3: Wire it up with JavaScript

Finally, add this function to your `<script>` tag. It fetches the data from the backend and builds out a list of stylized alert cards, reusing some of the CSS properties already present in your app to keep the design cohesive.

```javascript
// ── FETCH ALERTS DATA ──
async function loadAlertsData() {
  try {
    const response = await fetch('http://localhost:3000/api/alerts');
    const alerts = await response.json();
    
    const container = document.getElementById('alertsContainer');
    container.innerHTML = ''; // Clear existing
    
    let unreadCount = 0;

    alerts.forEach((alert, index) => {
      if (alert.isNew) unreadCount++;

      // Determine colors based on severity
      let icon = 'ℹ️';
      let bgColor = 'var(--surface2)';
      let iconBg = 'rgba(34,211,238,0.12)';
      
      if (alert.severity === 'high') {
        icon = '🚨';
        bgColor = alert.isNew ? 'rgba(248,113,113,0.05)' : 'var(--surface2)';
        iconBg = 'rgba(248,113,113,0.12)';
      } else if (alert.severity === 'med') {
        icon = '⚠️';
        bgColor = alert.isNew ? 'rgba(250,204,21,0.05)' : 'var(--surface2)';
        iconBg = 'rgba(250,204,21,0.12)';
      }

      const alertHTML = `
        <div style="display: flex; gap: 16px; padding: 18px 24px; border-bottom: 1px solid var(--border); background: ${bgColor}; transition: background 0.2s;">
          
          <div style="width: 40px; height: 40px; border-radius: 10px; background: ${iconBg}; display: flex; align-items: center; justify-content: center; font-size: 1.2rem; flex-shrink: 0;">
            ${icon}
          </div>
          
          <div style="flex: 1;">
            <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 4px;">
              <div style="display: flex; align-items: center; gap: 8px;">
                <h4 style="font-size: 0.9rem; font-weight: 600;">${alert.title}</h4>
                ${alert.isNew ? `<span style="background: var(--cyan); color: #000; font-size: 0.6rem; font-weight: 700; padding: 2px 6px; border-radius: 4px; text-transform: uppercase;">New</span>` : ''}
              </div>
              <span style="font-family: var(--mono); font-size: 0.7rem; color: var(--text3);">${alert.time}</span>
            </div>
            
            <p style="font-size: 0.8rem; color: var(--text2); line-height: 1.5; margin-bottom: 8px;">
              ${alert.message}
            </p>
            
            <div style="font-family: var(--mono); font-size: 0.65rem; color: var(--text3); text-transform: uppercase; letter-spacing: 0.05em;">
              Type: ${alert.type} | ID: ${alert.id}
            </div>
          </div>
          
        </div>
      `;
      
      container.innerHTML += alertHTML;
    });

    // Update the notification badge in the sidebar
    const badge = document.getElementById('alertBadge');
    if (badge) {
      if (unreadCount > 0) {
        badge.textContent = unreadCount;
        badge.style.display = 'inline-block';
      } else {
        badge.style.display = 'none';
      }
    }

  } catch (error) {
    console.error("Failed to load alerts:", error);
  }
}

// Helper function to simulate marking all as read
function markAllRead() {
  const badge = document.getElementById('alertBadge');
  if (badge) badge.style.display = 'none';
  
  // Reload the data (in a real app, you'd send a PUT request to the backend first)
  loadAlertsData(); 
}
