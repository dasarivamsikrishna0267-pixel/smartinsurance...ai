// --- AUDIT LOGS API ROUTES ---

// 1. GET: Fetch all audit logs (for the Audit Logs dashboard page)
app.get('/api/audit-logs', (req, res) => {
  // Always sort logs chronologically, newest first
  const sortedLogs = [...auditLogs].sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
  res.json(sortedLogs);
});

// 2. POST: Create a new audit log
// Note: In a real app, this route might be internal only, meaning the frontend 
// doesn't call it directly. Instead, your other backend routes call a log function.
app.post('/api/audit-logs', (req, res) => {
  const newLog = req.body;
  
  if (!newLog.action || !newLog.resource) {
    return res.status(400).json({ error: "Action and resource are required for an audit log." });
  }

  // Generate a sequential or UUID log ID
  newLog.id = `LOG-${Math.floor(Math.random() * 90000 + 10000)}`;
  
  // ALWAYS set the timestamp on the server side so the user cannot fake the time
  newLog.timestamp = new Date().toISOString();
  
  // If no user is provided, default to 'System'
  newLog.userId = newLog.userId || "System";
  newLog.userName = newLog.userName || "System Process";
  
  auditLogs.push(newLog);
  
  res.status(201).json({ message: "Audit log recorded successfully", log: newLog });
});
