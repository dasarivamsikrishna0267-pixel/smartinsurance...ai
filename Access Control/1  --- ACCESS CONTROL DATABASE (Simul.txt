// --- ACCESS CONTROL API ROUTES ---

// 1. GET: Fetch all roles and their permissions
app.get('/api/access/roles', (req, res) => {
  res.json(rolesAndPermissions);
});

// 2. GET: Fetch all internal system users
app.get('/api/access/users', (req, res) => {
  res.json(systemUsers);
});

// 3. POST: Invite/Add a new internal user to the dashboard
app.post('/api/access/users', (req, res) => {
  const newUser = req.body;
  
  if (!newUser.name || !newUser.email || !newUser.role) {
    return res.status(400).json({ error: "Name, email, and role are required." });
  }

  // Ensure the role actually exists in our system
  if (!rolesAndPermissions[newUser.role]) {
    return res.status(400).json({ error: "Invalid role specified." });
  }

  newUser.id = `USR-${Math.floor(Math.random() * 900 + 100)}`;
  newUser.status = newUser.status || "Pending Invite"; // They need to set a password first
  newUser.lastLogin = null;
  
  systemUsers.push(newUser);
  
  res.status(201).json({ message: "User invited successfully", user: newUser });
});

// 4. PUT: Change a user's role or status (e.g., Promoting someone or Suspending an account)
app.put('/api/access/users/:id', (req, res) => {
  const userId = req.params.id;
  const userIndex = systemUsers.findIndex(u => u.id === userId);
  
  if (userIndex === -1) {
    return res.status(404).json({ error: "System user not found." });
  }

  // If changing role, verify it's a valid role
  if (req.body.role && !rolesAndPermissions[req.body.role]) {
    return res.status(400).json({ error: "Invalid role specified." });
  }
  
  systemUsers[userIndex] = { ...systemUsers[userIndex], ...req.body, id: userId };
  
  res.json({ message: "User access updated successfully", user: systemUsers[userIndex] });
});

// 5. DELETE: Revoke access completely (Remove user)
app.delete('/api/access/users/:id', (req, res) => {
  const userId = req.params.id;
  const initialLength = systemUsers.length;
  
  // Super Admin safety check: prevent deleting the last Super Admin (Sneha Rao)
  const userToDelete = systemUsers.find(u => u.id === userId);
  if (userToDelete && userToDelete.role === "Super Admin") {
    const adminCount = systemUsers.filter(u => u.role === "Super Admin").length;
    if (adminCount <= 1) {
      return res.status(403).json({ error: "Cannot delete the last Super Admin. System requires at least one." });
    }
  }

  systemUsers = systemUsers.filter(u => u.id !== userId);
  
  if (systemUsers.length === initialLength) {
    return res.status(404).json({ error: "System user not found." });
  }
  
  res.json({ message: "User access revoked and account deleted." });
});
