// ── LOAD FRAUD CENTER ──
async function loadFraudCenter() {
  // 1. Show the main dashboard/claims page container
  showPage('dashPage');

  // 2. Change the page headers
  document.querySelector('.panel-head h3').textContent = 'Fraud Intelligence Center';
  document.querySelector('.panel-head p').textContent = 'High-risk claims flagged by AI forensics, metadata analysis, and rule engines';

  try {
    // 3. Fetch the specific Fraud Stats from our backend
    const statsResponse = await fetch('http://localhost:3000/api/fraud-stats');
    const stats = await statsResponse.json();

    // Update the sidebar badge
    document.getElementById('fraudBadge').textContent = stats.totalFlagged;

    // 4. Update the Top KPI Cards
    const kpiVals = document.querySelectorAll('.kpi-val');
    const kpiLabels = document.querySelectorAll('.kpi-label');
    const kpiIcons = document.querySelectorAll('.kpi-icon');
    
    // Card 1: Total Flagged
    kpiVals[0].textContent = stats.totalFlagged;
    kpiLabels[0].textContent = "Active Fraud Flags";
    kpiIcons[0].textContent = "🛡️";
    kpiIcons[0].className = "kpi-icon red";
    
    // Card 2: Capital at Risk
    kpiVals[1].textContent = stats.totalExposedValue;
    kpiLabels[1].textContent = "Capital at Risk";
    kpiIcons[1].textContent = "⚠️";
    kpiIcons[1].className = "kpi-icon yellow";
    
    // Card 3: Avg Fraud Score
    kpiVals[2].textContent = stats.averageFraudScore;
    kpiLabels[2].textContent = "Avg. Fraud Score";
    kpiIcons[2].textContent = "📈";
    kpiIcons[2].className = "kpi-icon red";
    
    // Card 4: Critical Alerts
    kpiVals[3].textContent = stats.criticalAlerts;
    kpiLabels[3].textContent = "Critical (80%+ Score)";
    kpiIcons[3].textContent = "🚨";
    kpiIcons[3].className = "kpi-icon red";

    // 5. Reuse our existing table function to load ONLY "Flagged" claims!
    loadAllClaims('', 'Flagged');

  } catch (error) {
    console.error("Error loading fraud center data:", error);
  }
}
