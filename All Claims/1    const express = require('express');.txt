// ==========================================
// API ENDPOINTS: ALL CLAIMS
// ==========================================

// 1. GET ALL CLAIMS (With optional filtering & searching)
app.get('/api/claims', (req, res) => {
  const { status, search } = req.query;
  let results = [...claimsDatabase];

  // Filter by status (e.g., /api/claims?status=Approved)
  if (status) {
    results = results.filter(claim => claim.status.toLowerCase() === status.toLowerCase());
  }

  // Filter by search query (e.g., /api/claims?search=Arjun)
  if (search) {
    const query = search.toLowerCase();
    results = results.filter(claim => 
      claim.customer.toLowerCase().includes(query) || 
      claim.id.toLowerCase().includes(query)
    );
  }

  res.status(200).json(results);
});

// 2. GET SINGLE CLAIM DETAILS
app.get('/api/claims/:id', (req, res) => {
  const claimId = req.params.id;
  const claim = claimsDatabase.find(c => c.id === claimId);
  
  if (!claim) {
    return res.status(404).json({ error: "Claim not found" });
  }
  
  res.status(200).json(claim);
});

// 3. UPDATE CLAIM STATUS (Approve, Reject, Hold)
app.patch('/api/claims/:id/status', (req, res) => {
  const claimId = req.params.id;
  const { status } = req.body;

  const claimIndex = claimsDatabase.findIndex(c => c.id === claimId);
  
  if (claimIndex === -1) {
    return res.status(404).json({ error: "Claim not found" });
  }

  // Update the status in the database
  claimsDatabase[claimIndex].status = status;

  console.log(`🔄 Claim ${claimId} updated to: ${status}`);
  res.status(200).json({ message: "Status updated successfully", claim: claimsDatabase[claimIndex] });
});

const PORT = 3000;
app.listen(PORT, () => console.log(`🚀 Claims API running on http://localhost:${PORT}`));
