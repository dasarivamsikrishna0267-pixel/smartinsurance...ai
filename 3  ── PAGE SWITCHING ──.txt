// ── PAGE SWITCHING ──
function showPage(pageId) {
  document.getElementById('dashPage').style.display = 'none';
  document.getElementById('uploadPage').style.display = 'none';
  document.getElementById(pageId).style.display = pageId === 'uploadPage' ? 'flex' : 'block';
  if (pageId === 'uploadPage') document.getElementById(pageId).style.flexDirection = 'column';
}

// Sidebar active state
function setActive(el) {
  document.querySelectorAll('.sb-item').forEach(i => i.classList.remove('active'));
  el.classList.add('active');
}

// Dashboard sidebar items default
document.querySelectorAll('.sb-item').forEach(item => {
  if (!item.getAttribute('onclick')) {
    item.addEventListener('click', function() { setActive(this); showPage('dashPage'); });
  }
});

// Drag & drop
dropZone.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('dragover'); });
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('dragover'));
dropZone.addEventListener('drop', e => {
  e.preventDefault(); dropZone.classList.remove('dragover');
  handleFiles(Array.from(e.dataTransfer.files).filter(f => f.type.startsWith('image/')));
});

fileInput.addEventListener('change', () => {
  handleFiles(Array.from(fileInput.files));
  fileInput.value = '';
});

function handleFiles(files) {
  if (!files.length) return;
  uploadedFiles = [...uploadedFiles, ...files];
  photoCount.textContent = uploadedFiles.length;
  thumbRow.style.display = 'block';

  // Update drop zone appearance
  document.getElementById('dzIcon').textContent = '✅';
  document.getElementById('dzTitle').textContent = `${uploadedFiles.length} photo${uploadedFiles.length>1?'s':''} uploaded`;
  document.getElementById('dzSub').textContent = 'Click or drag to add more photos';

  // Rebuild thumbnails
  renderThumbnails();

  // Show first new file in preview and run analysis
  showPreview(uploadedFiles[uploadedFiles.length - 1]);
}

function renderThumbnails() {
  photoThumbs.innerHTML = '';
  uploadedFiles.forEach((file, idx) => {
    const url = URL.createObjectURL(file);
    const wrap = document.createElement('div');
    wrap.className = 'thumb-wrap' + (idx === uploadedFiles.length-1 ? ' selected' : '');
    wrap.innerHTML = `<img src="${url}" alt="photo ${idx+1}"><div class="thumb-remove" onclick="removePhoto(${idx},event)">✕</div>`;
    wrap.onclick = (e) => { if (!e.target.classList.contains('thumb-remove')) { document.querySelectorAll('.thumb-wrap').forEach(t=>t.classList.remove('selected')); wrap.classList.add('selected'); showPreview(uploadedFiles[idx]); }};
    photoThumbs.appendChild(wrap);
  });
  const addMore = document.createElement('div');
  addMore.className = 'add-more-thumb';
  addMore.textContent = '+';
  addMore.onclick = () => fileInput.click();
  photoThumbs.appendChild(addMore);
}

function removePhoto(idx, e) {
  e.stopPropagation();
  uploadedFiles.splice(idx, 1);
  photoCount.textContent = uploadedFiles.length;
  if (uploadedFiles.length === 0) {
    thumbRow.style.display = 'none';
    previewImg.style.display = 'none';
    previewPlaceholder.style.display = 'flex';
    previewPlaceholder.style.flexDirection = 'column';
    previewPlaceholder.style.alignItems = 'center';
    scanOverlay.classList.remove('active');
    processingBadge.style.display = 'none';
    doneBadge.style.display = 'none';
    resetAIResults();
    document.getElementById('dzIcon').textContent = '🚗';
    document.getElementById('dzTitle').textContent = 'Drop vehicle photos here';
    document.getElementById('dzSub').textContent = 'Upload front, rear, side, and damage close-ups\nfor the most accurate AI assessment';
  } else {
    renderThumbnails();
    showPreview(uploadedFiles[Math.min(idx, uploadedFiles.length-1)]);
  }
}

function showPreview(file) {
  const url = URL.createObjectURL(file);
  previewPlaceholder.style.display = 'none';
  previewImg.src = url;
  previewImg.style.display = 'block';
  doneBadge.style.display = 'none';
  document.getElementById('damageBoxes').innerHTML = '';
  runAIAnalysis();
}

  submitBtn.disabled = true;
  analysisComplete = false;
  setStep(1);
}

function finishAnalysis(scenario) {

  scanOverlay.classList.remove('active');
  processingBadge.style.display = 'none';
  doneBadge.style.display = 'flex';

  // Place damage boxes
  const boxContainer = document.getElementById('damageBoxes');
  boxContainer.innerHTML = '';
  scenario.boxes.forEach((b, i) => {
    const box = document.createElement('div');
    box.className = 'dmg-box';
    box.style.cssText = `top:${b.top};left:${b.left};width:${b.w};height:${b.h}`;
    box.innerHTML = `<span class="dmg-label">${b.lbl}</span>`;
    boxContainer.appendChild(box);
    setTimeout(() => box.classList.add('show'), i * 200);
  });

  // Update AI chip
  aiChip.className = 'ai-status-chip chip-done';
  aiChip.textContent = '✓ Analysis Complete';

  // Fraud colour
  const fc = scenario.fraudScore < 25 ? 'var(--green)' : scenario.fraudScore < 60 ? 'var(--yellow)' : 'var(--red)';
  const fl = scenario.fraudScore < 25 ? 'Low' : scenario.fraudScore < 60 ? 'Medium' : 'High';

  // Render results
  aiResultsBody.innerHTML = `
    <div style="display:flex;flex-direction:column;gap:12px;">
      <div>
        ${scenario.damages.map(d => `
          <div class="dmg-result-bar">
            <div class="drb-row">
              <span>${d.label}</span>
              <span style="color:${d.color};font-weight:700;font-family:var(--mono)">${d.pct}% — ${d.severity}</span>
            </div>

// Step indicator
function setStep(n) {
  for (let i = 1; i <= 4; i++) {
    const el = document.getElementById('step' + i);
    if (!el) continue;
    el.classList.remove('active', 'done');
    if (i < n) el.classList.add('done');
    else if (i === n) el.classList.add('active');
  }
}
